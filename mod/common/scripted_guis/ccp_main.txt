scripted_gui = {

	ccp_hotkey_listener = {
		context_type = player_context
		parent_window_token = top_bar
		window_name = "ccp_hotkey_listener_window"

		visible = {
			always = yes
		}

		effects = {
			ccp_hotkey_1_click = {
				set_temp_variable = { ccp_hotkey_pressed = 1 }
				CCP_handle_hotkey = yes
			}
			ccp_hotkey_2_click = {
				set_temp_variable = { ccp_hotkey_pressed = 2 }
				CCP_handle_hotkey = yes
			}
			ccp_hotkey_3_click = {
				set_temp_variable = { ccp_hotkey_pressed = 3 }
				CCP_handle_hotkey = yes
			}
			ccp_hotkey_4_click = {
				set_temp_variable = { ccp_hotkey_pressed = 4 }
				CCP_handle_hotkey = yes
			}
		}
	}

	ccp_main_panel = {
		context_type = player_context
		parent_window_token = top_bar
		window_name = "ccp_main_window"

		visible = {
			check_variable = { ccp_panel_open = 1 }
		}

		dirty = ccp_ui_revision

		effects = {
			ccp_close_click = {
				CCP_close_panel = yes
			}
		}
	}
}
