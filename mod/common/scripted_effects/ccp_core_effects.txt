CCP_ensure_settings = {
	if = {
		limit = {
			check_variable = { ccp_settings_version < 1 }
		}
		set_variable = { ccp_settings_version = 1 }
		set_variable = { ccp_hotkey_id = 1 }
		set_variable = { ccp_confirm_destructive = 1 }
		set_variable = { ccp_section = 1 }
		set_variable = { ccp_equipment_category = 1 }
		set_variable = { ccp_equipment_selected = -1 }
		set_variable = { ccp_equipment_amount = 1000 }
	}
}

CCP_mark_dirty = {
	add_to_variable = { ccp_ui_revision = 1 }
}

CCP_open_panel = {
	CCP_ensure_settings = yes
	set_variable = { ccp_panel_open = 1 }
	CCP_mark_dirty = yes
}

CCP_close_panel = {
	set_variable = { ccp_panel_open = 0 }
	CCP_mark_dirty = yes
}

CCP_toggle_panel = {
	CCP_ensure_settings = yes
	if = {
		limit = {
			check_variable = { ccp_panel_open = 1 }
		}
		CCP_close_panel = yes
	}
	else = {
		CCP_open_panel = yes
	}
}
